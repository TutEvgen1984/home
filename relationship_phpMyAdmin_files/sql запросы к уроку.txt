==== Запрос для таблиц Фильмы и Жанры ====

SELECT

films.name AS 'Название',
genre.genre AS 'Жанр'

FROM films_genre
LEFT JOIN films ON (films.id_film = films_genre.id_film)
LEFT JOIN genre ON (genre.id_genre = films_genre.id_genre)

==== Запрос для БД фильмы ====

SELECT

films.name_ru AS 'Название на русском',
films.name_en  AS 'Название на английском',
films.year AS 'Год',
genre.genre AS 'Жанр',
country.country AS 'Страна',

(SELECT GROUP_CONCAT(actors.name SEPARATOR ', ') FROM actors, actors_films WHERE 
 actors.id_actor=actors_films.id_actor and 
 actors_films.id_film=films.id_film) AS 'Актеры'

FROM films
LEFT JOIN genre ON films.id_genre=genre.id_genre
LEFT JOIN country ON films.id_country=country.id_country
